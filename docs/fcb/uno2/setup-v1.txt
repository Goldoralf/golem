// ── 7c. stomp 10 : gestes complets (espèce « trigger ») ─────────────────────
//     la bascule CC34 est reconstruite par variable booléenne
TRIGGER_CLICK Stomp10 =
{
  if ($s10)
  {
    SendMidi Golem CtrlChange 34 0
  }
  else
  {
    SendMidi Golem CtrlChange 34 127
  }
  $s10 = !$s10
  SendMidi Golem NoteOn 9 100
}
TRIGGER_RELEASE Stomp10 = SendMidi Golem NoteOff 9 0
